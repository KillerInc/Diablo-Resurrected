cmake_minimum_required(VERSION 3.16)
project(diablo_reincarnated_renderer_foundation LANGUAGES CXX)
find_package(Threads REQUIRED)

add_library(reincarnated_renderer_foundation STATIC renderer_switch.cpp sprite_layout.cpp asset_provider.cpp)
target_link_libraries(reincarnated_renderer_foundation PUBLIC Threads::Threads)
target_include_directories(reincarnated_renderer_foundation PUBLIC .)
target_compile_features(reincarnated_renderer_foundation PUBLIC cxx_std_17)

include(CTest)
if(BUILD_TESTING)
    # These tests use assert, including expressions that exercise behavior.
    # Keep them active in the Release configuration used by Windows CI.
    if(MSVC)
        add_compile_options(/UNDEBUG)
    else()
        add_compile_options(-UNDEBUG)
    endif()
    add_executable(reincarnated_renderer_foundation_test renderer_foundation_test.cpp)
    target_link_libraries(reincarnated_renderer_foundation_test PRIVATE reincarnated_renderer_foundation)
    add_test(NAME renderer_switch_and_hd_sprite_layout COMMAND reincarnated_renderer_foundation_test)
    add_executable(reincarnated_scene_commands_test scene_commands_test.cpp)
    target_link_libraries(reincarnated_scene_commands_test PRIVATE reincarnated_renderer_foundation)
    add_test(NAME renderer_depth_ordering_and_zoom_mapping COMMAND reincarnated_scene_commands_test)
    add_executable(reincarnated_dds_decode_test dds_decode_test.cpp)
    add_test(NAME hd_dds_decode_and_invalid_asset_fallback COMMAND reincarnated_dds_decode_test)
    add_executable(reincarnated_asset_provider_test asset_provider_test.cpp)
    target_link_libraries(reincarnated_asset_provider_test PRIVATE reincarnated_renderer_foundation)
    add_test(NAME hd_loose_files_and_archive_provider_fallback COMMAND reincarnated_asset_provider_test)
endif()

if(BUILD_TESTING)
    add_executable(reincarnated_performance_samples_test performance_samples_test.cpp)
    add_test(NAME performance_percentiles_and_hitches COMMAND reincarnated_performance_samples_test)
endif()
